function currentSlot(){
  const h=new Date().getHours();
  if(h>=5&&h<12)return'morning';
  if(h>=12&&h<17)return'afternoon';
  if(h>=17&&h<21)return'evening';
  return'night';
}

function asthmaControl(items){
  const last7=items.filter(x=>daysSince(x.date)<=7);
  const rel=last7.reduce((s,x)=>s+x.reliever,0);
  const sym=last7.filter(x=>x.symptoms.length&&!x.symptoms.includes('None')).length;
  if(!last7.length)return{label:'No data',c:'var(--ink-dim)',rel:0};
  if(rel<=2&&sym<=1)return{label:'Well controlled',c:'var(--green)',rel};
  if(rel<=6&&sym<=3)return{label:'Partly controlled',c:'var(--amber)',rel};
  return{label:'Needs attention',c:'var(--coral)',rel};
}

  const items=load(DB.asthma).slice().sort((a,b)=>b.date.localeCompare(a.date));

  document.getElementById('as-pf').innerHTML=items[0]?.pf?`${items[0].pf}<small> L/min</small>`:'—';
  const st=document.getElementById('as-status');st.textContent=ctrl.label;st.style.color=ctrl.c;
  document.getElementById('as-reliever').textContent=ctrl.rel;
  const list=document.getElementById('as-list');

/* ---- Export action plan as PDF (recreates the NAC Australia template) ---- */
function exportAsthmaPlanPDF(){
  const p=loadObj(DB.asthmaPlan);
  if(!(p.doctor||p.preventer||p.reliever||p.z1s||p.z2s||p.z3s)){toast('Fill in your action plan first');return;}
  loadJsPDF(()=>buildAsthmaPDF(p),()=>toast('Could not load PDF tool (check connection)'));
}
function loadJsPDF(ok,fail){
  if(window.jspdf&&window.jspdf.jsPDF){ok();return;}
  const s=document.createElement('script');
  s.src='https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js';
  s.onload=ok; s.onerror=fail||(()=>{});
  document.head.appendChild(s);
}
function buildAsthmaPDF(p){
  const {jsPDF}=window.jspdf;
  const doc=new jsPDF({unit:'mm',format:'a4'});
  const s=loadObj(DB.settings), em=loadObj(DB.emergency);
  const name=s.name||em.name||'';
  const M=15, W=210, R=195, CW=180;
  const GREEN=[46,155,87], AMBER=[230,140,12], RED=[214,68,58], INK=[28,46,36], MUTE=[110,132,118];
  const tint=(c,f)=>c.map(v=>Math.round(v*f+255*(1-f)));
  let y=0;
  const ensure=h=>{if(y+h>284){doc.addPage();y=18;}};

  // header band
  doc.setFillColor(8,163,82); doc.rect(0,0,W,26,'F');
  doc.setTextColor(255,255,255);
  doc.setFont('helvetica','bold').setFontSize(20); doc.text('Asthma Action Plan',M,15);
  doc.setFont('helvetica','normal').setFontSize(8.5); doc.text('A personal plan — to be completed with your doctor',M,21);
  doc.setFont('helvetica','bold').setFontSize(13); doc.text('WellNest',R,14,{align:'right'});
  doc.setFont('helvetica','normal').setFontSize(7.5); doc.text('Health, centered.',R,19,{align:'right'});
  y=34;

  // detail fields row
  const field=(label,val,x,w)=>{
    doc.setTextColor(...MUTE).setFont('helvetica','bold').setFontSize(7);
    doc.text(label.toUpperCase(),x,y);
    doc.setTextColor(...INK).setFont('helvetica','normal').setFontSize(10);
    doc.text(doc.splitTextToSize(val||'—',w),x,y+5);
    doc.setDrawColor(225,232,228); doc.line(x,y+7.5,x+w-6,y+7.5);
  };
  field('Name',name,M,90); field('Date',new Date().toLocaleDateString('en-AU'),M+95,85);
  y+=15;
  field('Completed with',p.doctor,M,60); field('Next review',p.review?fmtDate(p.review):'',M+62,55);
  field('Best peak flow',p.best?p.best+' L/min':'',M+122,58);
  y+=16;

  // usual medicines box
  ensure(34);
  doc.setFillColor(...tint(GREEN,.07)); doc.setDrawColor(...tint(GREEN,.5));
  const medLines=[['Preventer (daily)',p.preventer],['Reliever (as needed)',p.reliever],['Other medicines',p.other]].filter(m=>m[1]);
  const medH=12+Math.max(medLines.length,1)*7;
  doc.roundedRect(M,y,CW,medH,2,2,'FD');
  doc.setTextColor(...INK).setFont('helvetica','bold').setFontSize(11); doc.text('My usual asthma medicines (when well)',M+4,y+7);
  let my=y+14;
  doc.setFontSize(9.5);
  if(medLines.length)medLines.forEach(m=>{
    doc.setFont('helvetica','bold').setTextColor(...MUTE); doc.text(m[0]+':',M+4,my);
    doc.setFont('helvetica','normal').setTextColor(...INK);
    doc.text(doc.splitTextToSize(m[1],CW-50),M+48,my); my+=7;
  });
  else {doc.setFont('helvetica','italic').setTextColor(...MUTE); doc.text('Not recorded',M+4,my);}

function buildMedsPDF(items){
  const {jsPDF}=window.jspdf;
  const doc=new jsPDF({unit:'mm',format:'a4'});
  const s=loadObj(DB.settings), em=loadObj(DB.emergency);
  const name=s.name||em.name||'';
  const M=15, W=210, R=195, CW=180;
  const TEAL=[12,160,143], INK=[28,46,36], MUTE=[110,132,118];
  const tint=(c,f)=>c.map(v=>Math.round(v*f+255*(1-f)));
  let y=0;
  const ensure=h=>{if(y+h>284){doc.addPage();y=18;drawTableHead();}};

  // header band
  doc.setFillColor(...TEAL); doc.rect(0,0,W,26,'F');
  doc.setTextColor(255,255,255);
  doc.setFont('helvetica','bold').setFontSize(20); doc.text('Medication List',M,15);
  doc.setFont('helvetica','normal').setFontSize(8.5); doc.text('A current record of medications and schedule',M,21);
  doc.setFont('helvetica','bold').setFontSize(13); doc.text('WellNest',R,14,{align:'right'});
  doc.setFont('helvetica','normal').setFontSize(7.5); doc.text('Health, centered.',R,19,{align:'right'});
  y=34;

  // detail fields
  const field=(label,val,x,w)=>{
    doc.setTextColor(...MUTE).setFont('helvetica','bold').setFontSize(7);
    doc.text(label.toUpperCase(),x,y);
    doc.setTextColor(...INK).setFont('helvetica','normal').setFontSize(10);
    doc.text(doc.splitTextToSize(val||'—',w),x,y+5);
    doc.setDrawColor(225,232,228); doc.line(x,y+7.5,x+w-6,y+7.5);
  };
  field('Name',name,M,90); field('Date',new Date().toLocaleDateString('en-AU'),M+95,85);
  y+=16;

  // column layout
  const C1=M+3, C2=M+60, C3=M+86, C4=M+124;
  const W1=54, W2=24, W3=36, W4=54;
  function drawTableHead(){
    doc.setFillColor(...tint(TEAL,.14)); doc.roundedRect(M,y,CW,9,1.5,1.5,'F');
    doc.setTextColor(...TEAL).setFont('helvetica','bold').setFontSize(7.5);
    doc.text('MEDICATION',C1,y+6); doc.text('DOSE',C2,y+6);
    doc.text('FREQUENCY',C3,y+6); doc.text('TIME(S)',C4,y+6);
    y+=12;
  }
  drawTableHead();

  // rows
  items.forEach((m,i)=>{
    const nameLines=doc.splitTextToSize(m.name||'—',W1);
    const doseLines=doc.splitTextToSize(m.dose||'—',W2);
    const freqLines=doc.splitTextToSize(m.freq||'—',W3);
    const timeLines=doc.splitTextToSize(m.time||'—',W4);
    const maxLines=Math.max(nameLines.length,doseLines.length,freqLines.length,timeLines.length);
    const sub=[m.for?'For: '+m.for:'',m.notes||''].filter(Boolean).join('   ·   ');
    const subLines=sub?doc.splitTextToSize(sub,CW-6):[];
    const rowH=maxLines*4.6+(subLines.length?subLines.length*3.8+2:0)+5;
    ensure(rowH+1);
    if(i%2===0){doc.setFillColor(247,251,250);doc.rect(M,y-2,CW,rowH,'F');}
    doc.setTextColor(...INK).setFont('helvetica','bold').setFontSize(9.5); doc.text(nameLines,C1,y+3);
    doc.setFont('helvetica','normal').setFontSize(9);
    doc.text(doseLines,C2,y+3); doc.text(freqLines,C3,y+3); doc.text(timeLines,C4,y+3);
    if(subLines.length){doc.setTextColor(...MUTE).setFont('helvetica','italic').setFontSize(7.5);
      doc.text(subLines,C1,y+3+maxLines*4.6);}
    y+=rowH;
    doc.setDrawColor(232,238,235); doc.line(M,y-2,M+CW,y-2);
  });
  y+=6;

  // summary
  ensure(8);
  doc.setTextColor(...MUTE).setFont('helvetica','normal').setFontSize(8.5);
  doc.text(`${items.length} medication${items.length!==1?'s':''} recorded`,M,y);

  // footer (last page)
  const foot='This is a personal record of medications generated by WellNest and does not replace professional medical advice. '+
    'Always confirm doses and schedules with your doctor or pharmacist. In an emergency call 000.   '+
    'Generated by WellNest · '+new Date().toLocaleDateString('en-AU');
  doc.setDrawColor(225,232,228); doc.line(M,287,R,287);
  doc.setTextColor(...MUTE).setFont('helvetica','normal').setFontSize(6.8);
  doc.text(doc.splitTextToSize(foot,CW),M,290);

  doc.save(`medications-${name?name.replace(/\s+/g,'-').toLowerCase()+'-':''}${today()}.pdf`);
  toast('Medications PDF exported');
}

function saveExpense(id){
  const date=document.getElementById('exp-date').value;
  const amount=parseFloat(document.getElementById('exp-amt').value)||0;
  const category=document.getElementById('exp-cat').value;
  const description=document.getElementById('exp-desc').value;
  const notes=document.getElementById('exp-notes').value;
  if(!date||!amount||amount<=0){toast('Fill in date and amount');return;}
  const items=load(DB.expenses);
  if(id){
    const idx=items.findIndex(x=>x.id===id);
    if(idx>=0)items[idx]={...items[idx],date,amount,category,description,notes};
  }else{
    items.push({id:uid(),date,amount,category,description,notes});
  }
  save(DB.expenses,items);closeModal();toast('Expense saved');renderAll();
}
function delExpense(id){
  if(confirm('Delete this expense?')){
    save(DB.expenses,load(DB.expenses).filter(x=>x.id!==id));
    closeModal();toast('Expense deleted');renderAll();
  }
}
function renderExpenses(){
  const list=document.getElementById('exp-list');
  const totalEl=document.getElementById('exp-total');
  const monthEl=document.getElementById('exp-month');
  if(!list||!totalEl||!monthEl)return;
  
  const items=load(DB.expenses).sort((a,b)=>new Date(b.date)-new Date(a.date));
  const now=new Date();
  const thisMonth=items.filter(e=>{
    const d=new Date(e.date);
    return d.getFullYear()===now.getFullYear()&&d.getMonth()===now.getMonth();
  });
  
  const total=items.reduce((s,e)=>s+e.amount,0);
  const monthTotal=thisMonth.reduce((s,e)=>s+e.amount,0);
  
  totalEl.textContent='$'+total.toFixed(2);
  monthEl.textContent='$'+monthTotal.toFixed(2);
  
  if(!items.length){list.innerHTML='<div style="text-align:center;color:var(--ink-dim);padding:40px 20px">No expenses tracked yet. Click "+ Add expense" to get started.</div>';return;}
  
  const catColor={
    'Doctor visit':'var(--coral)',
    'Medication':'var(--teal)',
    'Hospital/ER':'var(--coral)',
    'Mental health':'var(--violet)',
    'Lab/imaging':'var(--amber)',
    'Equipment':'var(--sky)',
    'Insurance':'var(--green)',
    'Other':'var(--ink-dim)'
  };
  
  let curr='';
  list.innerHTML=items.map(e=>{
    const d=fmtDate(e.date);
    const h=curr!==d?`<div style="font-size:13px;font-weight:700;color:var(--ink-dim);margin:20px 0 12px 0;text-transform:uppercase;letter-spacing:0.5px">${d}</div>`:'';
    curr=d;
    const color=catColor[e.category]||'var(--ink-dim)';
    return h+`<div class="row" style="margin-bottom:12px">
      <div style="flex:1">
        <div style="display:flex;align-items:center;gap:8px;margin-bottom:4px">
          <div style="width:8px;height:8px;border-radius:50%;background:${color}"></div>
          <div style="font-weight:600;font-size:14px">${esc(e.description)||e.category}</div>

      </div>
      <div style="text-align:right;margin-right:10px">
        <div style="font-weight:600;font-size:14px;color:var(--amber)">$${e.amount.toFixed(2)}</div>
      </div>
      <div class="icon-btn" onclick="openExpense('${e.id}')" style="color:var(--amber)">✎</div>
    </div>`;

}

function buildCycleDots(){
  const slides=document.querySelectorAll('#cycleWidget .cycle-slide');
  const dots=document.getElementById('cw-dots');
  dots.innerHTML=Array.from(slides).map((_,i)=>`<button class="cycle-dot" onclick="goCycleSlide(${i})"></button>`).join('');
}
function showCycleSlide(i){
  const slides=document.querySelectorAll('#cycleWidget .cycle-slide');
  const dots=document.querySelectorAll('#cycleWidget .cycle-dot');
  if(!slides.length)return;
  const prev=cwIndex;
  cwIndex=(i+slides.length)%slides.length;
  // arriving at the inspiration slide (0) from another slide → show a fresh quote
  if(cwIndex===0&&prev!==0){
    cwQuoteIndex=(cwQuoteIndex+1)%PAIN_QUOTES.length;
    setCycleQuote();
  }
  slides.forEach((s,n)=>{
    const accent=s.getAttribute('data-accent');
    s.style.setProperty('--sa',`var(${accent})`);
    s.classList.toggle('active',n===cwIndex);
  });
  dots.forEach((d,n)=>{
    d.classList.toggle('active',n===cwIndex);
    if(n===cwIndex){const accent=slides[cwIndex].getAttribute('data-accent');d.style.setProperty('--sa',`var(${accent})`);}
  });
}
function goCycleSlide(i){showCycleSlide(i);startCycle();}
function startCycle(){
  clearInterval(cwTimer);
  const dash=document.getElementById('page-dashboard');
  if(!dash||!dash.classList.contains('active'))return;
  cwTimer=setInterval(()=>showCycleSlide(cwIndex+1),6000);
}

function exportData(){
  const dump={exported:new Date().toISOString(),version:1};
  Object.entries(DB).forEach(([k,key])=>{const v=localStorage.getItem(key);if(v!=null)dump[key]=JSON.parse(v);});
  const blob=new Blob([JSON.stringify(dump,null,2)],{type:'application/json'});
  const a=document.createElement('a');a.href=URL.createObjectURL(blob);
  a.download=`wellnest-backup-${today()}.json`;a.click();URL.revokeObjectURL(a.href);
  toast('Data exported');
}
function importData(ev){
  const file=ev.target.files[0]; if(!file)return;
  const r=new FileReader();
  r.onload=()=>{try{
    const data=JSON.parse(r.result);
    let n=0;
    Object.keys(DB).forEach(k=>{if(data[k]!==undefined){save(DB[k],data[k]);n++;}});
    if(n){localStorage.setItem('wn-seeded','1');toast('Data imported');applyGreeting();renderAll();}
    else toast('No WellNest data found in file');
  }catch(e){toast('Could not read that file');}
  ev.target.value='';};
  r.readAsText(file);
}

// Replace with your Google OAuth 2.0 Client ID from https://console.cloud.google.com
const GOOGLE_CLIENT_ID='YOUR_GOOGLE_CLIENT_ID_HERE';
const googleAuthKey='wn-google-auth';

function initGoogleSignIn(){
  const box=document.getElementById('google-signin-box');
  const info=document.getElementById('google-user-info');
  if(!box||!info)return;
  if(!window.google){setTimeout(initGoogleSignIn,100);return;}
  const auth=loadObj(googleAuthKey);
  if(auth.name){
    box.style.display='none';
    info.style.display='block';
    document.getElementById('google-user-name').textContent=auth.name;
    document.getElementById('google-user-email').textContent=auth.email;
    if(auth.picture)document.getElementById('google-user-pic').src=auth.picture;
  }else{
    google.accounts.id.initialize({client_id:GOOGLE_CLIENT_ID,callback:googleCallback});
    google.accounts.id.renderButton(box,{
      theme:'outline',width:'100%',size:'large',text:'signin_with'
    });
  }
}
function googleCallback(response){
  try{
    const base64Url=response.credential.split('.')[1];
    const base64=base64Url.replace(/-/g,'+').replace(/_/g,'/');
    const decoded=JSON.parse(atob(base64));
    const auth={name:decoded.name,email:decoded.email,picture:decoded.picture,id_token:response.credential};
    save(googleAuthKey,auth);
    toast('Signed in with Google');
    renderSettings();
  }catch(e){console.error('Google auth error:',e);}
}
function googleSignOut(){
  localStorage.removeItem(googleAuthKey);
  toast('Signed out');
  renderSettings();
}

/* Configure these from your Stripe Dashboard.
   - Create Payment Links (Products → Payment Links) for each plan, set a 7-day free trial,
     and set the post-payment redirect to this app's URL with ?checkout=success appended.
   - Create a Customer Portal link (Settings → Billing → Customer portal) for subscription management.
   Publishable key / links are safe to expose; the SECRET key must NEVER appear in this file.
   Real subscription status must be confirmed server-side via Stripe webhooks — the local
   flag below is for UI display only and can be spoofed by the user. */
const STRIPE_CONFIG={
  monthly:{price:'$4.99 AUD',period:'/month',link:'https://buy.stripe.com/YOUR_MONTHLY_LINK'},
  yearly:{price:'$44.99 AUD',period:'/year',note:'Save 20% — 2 months free',link:'https://buy.stripe.com/YOUR_YEARLY_LINK'},
  trialDays:7,
  portalLink:'https://billing.stripe.com/p/login/YOUR_PORTAL_LINK'
};

function startCheckout(plan){
  const cfg=STRIPE_CONFIG[plan];
  if(!cfg||cfg.link.includes('YOUR_')){
    toast('Add your Stripe Payment Link first');return;
  }
  // attach the signed-in email so Stripe pre-fills & links the customer
  const auth=loadObj(googleAuthKey);
  let url=cfg.link;
  if(auth.email)url+=(url.includes('?')?'&':'?')+'prefilled_email='+encodeURIComponent(auth.email);
  window.location.href=url;
}
function openBillingPortal(){
  if(STRIPE_CONFIG.portalLink.includes('YOUR_')){toast('Add your Customer Portal link first');return;}
  window.open(STRIPE_CONFIG.portalLink,'_blank');
}
function checkCheckoutReturn(){
  const params=new URLSearchParams(window.location.search);
  if(params.get('checkout')==='success'){
    const b=loadObj(DB.billing); b.plan='premium'; b.since=today(); save(DB.billing,b);
    toast('Welcome to WellNest Premium');
    history.replaceState(null,'',window.location.pathname);
  }else if(params.get('checkout')==='cancel'){
    history.replaceState(null,'',window.location.pathname);
  }
}

      You're on <b style="color:var(--green)">WellNest Premium</b>${b.since?` since ${fmtDate(b.since)}`:''}.
      Thanks for supporting WellNest. Manage your plan, payment method or invoices in the billing portal.</div>
      <button class="btn primary" onclick="openBillingPortal()">Manage billing →</button>`;
    return;
  }
  const m=STRIPE_CONFIG.monthly, y=STRIPE_CONFIG.yearly;
  body.innerHTML=`<div style="font-size:13px;color:var(--ink-dim);margin-bottom:16px">
      Start with a ${STRIPE_CONFIG.trialDays}-day free trial. Cancel anytime.</div>
